# 🎯 Parking Lot System Design

Design an **Object-Oriented Parking Lot System** that manages multiple floors, different vehicle types, ticket generation, and payment processing.

<Callout type="tip" title="Real-World Analogy">
  A multi-level parking garage at a shopping mall — it has entry/exit gates, displays available spots per floor, issues tickets on entry, calculates fees on exit, and supports cars, motorcycles, and trucks.
</Callout>

---

## 📋 Requirements

### Functional
- Support multiple floors with different spot sizes (Compact, Regular, Large)
- Handle multiple vehicle types (Motorcycle, Car, Truck)
- Assign the nearest available spot to a vehicle
- Issue parking tickets on entry
- Calculate fees based on duration
- Track availability per floor

### Non-Functional
- Thread-safe for concurrent entry/exit
- Extensible for new vehicle types

---

## 🔍 Class Design

| Class | Responsibility |
|-------|---------------|
| `ParkingLot` | Singleton managing the entire lot |
| `ParkingFloor` | Manages spots on one floor |
| `ParkingSpot` | Individual spot (type + availability) |
| `Vehicle` | Abstract base for vehicle types |
| `ParkingTicket` | Records entry time, spot, vehicle |
| `PaymentProcessor` | Calculates and processes fees |

from datetime import datetime
from typing import Optional
import threading

class VehicleType(Enum):
    MOTORCYCLE = 1
    CAR = 2
    TRUCK = 3

class SpotSize(Enum):
    COMPACT = 1    # Motorcycle
    REGULAR = 2    # Car
    LARGE = 3      # Truck

SPOT_COMPATIBILITY = {
    VehicleType.MOTORCYCLE: [SpotSize.COMPACT, SpotSize.REGULAR, SpotSize.LARGE],
    VehicleType.CAR: [SpotSize.REGULAR, SpotSize.LARGE],
    VehicleType.TRUCK: [SpotSize.LARGE],
}

class Vehicle:
    def __init__(self, license_plate: str, v_type: VehicleType):
        self.license_plate = license_plate
        self.type = v_type

class ParkingSpot:
    def __init__(self, spot_id: str, size: SpotSize, floor: int):
        self.spot_id = spot_id
        self.size = size
        self.floor = floor
        self.vehicle: Optional[Vehicle] = None

    @property
    def is_available(self) -> bool:
        return self.vehicle is None

    def park(self, vehicle: Vehicle) -> bool:
        if not self.is_available:
            return False
        if self.size not in SPOT_COMPATIBILITY.get(vehicle.type, []):
            return False
        self.vehicle = vehicle
        return True

    def remove_vehicle(self) -> Optional[Vehicle]:
        v = self.vehicle
        self.vehicle = None
        return v

class ParkingTicket:
    _counter = 0

    def __init__(self, vehicle: Vehicle, spot: ParkingSpot):
        ParkingTicket._counter += 1
        self.ticket_id = f"T-{ParkingTicket._counter:04d}"
        self.vehicle = vehicle
        self.spot = spot
        self.entry_time = datetime.now()
        self.exit_time: Optional[datetime] = None
        self.paid = False

    def get_duration_hours(self) -> float:
        end = self.exit_time or datetime.now()
        return (end - self.entry_time).total_seconds() / 3600

class ParkingLot:
    _instance = None
    _lock = threading.Lock()

    def __new__(cls, *args, **kwargs):
        with cls._lock:
            if cls._instance is None:
                cls._instance = super().__new__(cls)
            return cls._instance

    def __init__(self, name: str, floors: int,
                 spots_per_floor: int):
        if hasattr(self, '_initialized'):
            return
        self.name = name
        self.spots: list[ParkingSpot] = []
        self.active_tickets: dict[str, ParkingTicket] = {}
        self._lock_spots = threading.Lock()

        for f in range(1, floors + 1):
            for s in range(1, spots_per_floor + 1):
                if s <= spots_per_floor // 4:
                    size = SpotSize.COMPACT
                elif s <= 3 * spots_per_floor // 4:
                    size = SpotSize.REGULAR
                else:
                    size = SpotSize.LARGE
                self.spots.append(
                    ParkingSpot(f"F{f}-S{s:03d}", size, f))
        self._initialized = True

    def park_vehicle(self, vehicle: Vehicle) -> Optional[ParkingTicket]:
        with self._lock_spots:
            compatible = SPOT_COMPATIBILITY[vehicle.type]
            for spot in self.spots:
                if spot.is_available and spot.size in compatible:
                    spot.park(vehicle)
                    ticket = ParkingTicket(vehicle, spot)
                    self.active_tickets[vehicle.license_plate] = ticket
                    print(f"✅ {vehicle.license_plate} → {spot.spot_id}")
                    return ticket
        print(f"❌ No spot for {vehicle.license_plate}")
        return None

    def exit_vehicle(self, license_plate: str,
                     hourly_rate: float = 5.0) -> Optional[float]:
        ticket = self.active_tickets.pop(license_plate, None)
        if not ticket:
            return None
        ticket.exit_time = datetime.now()
        ticket.spot.remove_vehicle()
        fee = max(1.0, ticket.get_duration_hours() * hourly_rate)
        print(f"🚗 {license_plate} exited. Fee: ${'$'}{fee:.2f}")
        return fee

    def get_availability(self) -> dict:
        avail = {}
        for spot in self.spots:
            key = f"Floor {spot.floor} - {spot.size.name}"
            avail[key] = avail.get(key, 0) + (1 if spot.is_available else 0)
        return avail

# Usage

car = Vehicle("ABC-1234", VehicleType.CAR)
ticket = lot.park_vehicle(car)
print(lot.get_availability())
lot.exit_vehicle("ABC-1234")`}

    private static volatile ParkingLot instance;
    private final List<ParkingSpot> spots;
    private final Map<String, ParkingTicket> activeTickets;
    private final ReentrantLock lock = new ReentrantLock();

    private ParkingLot(int floors, int spotsPerFloor) {
        spots = new ArrayList<>();
        activeTickets = new ConcurrentHashMap<>();
        for (int f = 1; f <= floors; f++) {
            for (int s = 1; s <= spotsPerFloor; s++) {
                SpotSize size = s <= spotsPerFloor / 4
                    ? SpotSize.COMPACT
                    : s <= 3 * spotsPerFloor / 4
                        ? SpotSize.REGULAR : SpotSize.LARGE;
                spots.add(new ParkingSpot(
                    "F%d-S%03d".formatted(f, s), size, f));
            }
        }
    }

    public static ParkingLot getInstance(int floors, int spots) {
        if (instance == null) {
            synchronized (ParkingLot.class) {
                if (instance == null)
                    instance = new ParkingLot(floors, spots);
            }
        }
        return instance;
    }

    public Optional<ParkingTicket> parkVehicle(Vehicle vehicle) {
        lock.lock();
        try {
            for (ParkingSpot spot : spots) {
                if (spot.isAvailable() && spot.canFit(vehicle)) {
                    spot.park(vehicle);
                    var ticket = new ParkingTicket(vehicle, spot);
                    activeTickets.put(
                        vehicle.getLicensePlate(), ticket);
                    return Optional.of(ticket);
                }
            }
            return Optional.empty();
        } finally {
            lock.unlock();
        }
    }

    public Optional<Double> exitVehicle(String plate, double rate) {
        var ticket = activeTickets.remove(plate);
        if (ticket == null) return Optional.empty();
        ticket.markExit();
        ticket.getSpot().removeVehicle();
        double fee = Math.max(1.0,
            ticket.getDurationHours() * rate);
        return Optional.of(fee);
    }
}`}

## 📊 Design Patterns Used

## 📝 Quick Quiz

**Q1: A truck arrives but all Large spots are taken. Should it park in two Regular spots?**

A) Yes, for space efficiency  B) No, violates the SpotSize constraint

**Answer: B** — Each vehicle parks in exactly one spot. A truck requires a Large spot. If none are available, it should be turned away. Splitting across spots adds dangerous complexity.
